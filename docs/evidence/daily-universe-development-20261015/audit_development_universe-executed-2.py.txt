def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(8*1024*1024), b''):
            h.update(block)
    return h.hexdigest()


def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8'))


def stable(value):
    return json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2, allow_nan=False)


def save(path, value):
    with path.open('x', encoding='utf-8', newline='\n') as stream:
        stream.write(stable(value)+'\n')

def catalog_snapshot():
    with duckdb.connect(str(root/'data/live/catalog.duckdb'), read_only=True,
                        config={'threads': 1, 'memory_limit': '256MB'}) as con:
        con.execute('BEGIN TRANSACTION')
        cursor = con.execute(query, [cutoff])
        names = [column[0] for column in cursor.description]
        entries = [dict(zip(names, row, strict=True)) for row in cursor.fetchall()]
    paths = sorted({(x['parquet_path'], x['parquet_hash']) for x in entries if x['kind'] == 'normalized'})
    digest = hashlib.sha256(json.dumps(paths, ensure_ascii=False, separators=(',', ':')).encode()).hexdigest()
    assert len(paths) == identity['normalized_objects_verified']
    assert digest == identity['normalized_path_hash_list_sha256'], 'Frozen master catalog has changed'
    return {x['object_id']: x for x in entries if start <= x['source_date'] <= end}

def authenticated_frame(entry):
    stored = store.verify(Path(entry['parquet_path']).parent)
    assert stored.provider == 'J-Quants' and stored.source_endpoint == '/equities/master'
    assert stored.dataset == DatasetName.SECURITY_MASTER
    for key in _CATALOG_FIELDS:
        expected, observed = getattr(stored, key), entry[key]
        if key in ('parquet_path', 'manifest_path'):
            assert Path(expected).resolve() == Path(observed).resolve(), key
        else:
            assert expected == observed, key
    frame = pd.read_parquet(stored.parquet_path)
    assert set(_META).issubset(frame.columns)
    for key in _META:
        if key == 'source_record_hash':
            assert frame[key].notna().all()
            assert frame[key].astype(str).str.fullmatch('[0-9a-f]{64}').all()
        else:
            expected = ENDPOINT_SCHEMAS[DatasetName.SECURITY_MASTER].schema_version if key == 'schema_version' else getattr(stored, key)
            assert frame[key].eq(expected).all(), key
    return stored, frame


mapping = {
    'company_name': 'CoName', 'company_name_en': 'CoNameEn',
    'sector_17_code': 'S17', 'sector_17_name': 'S17Nm',
    'sector_33_code': 'S33', 'sector_33_name': 'S33Nm',

        assert catalog_snapshot() == catalog

            assert sha(root/path) == expected
        checkpoint_path = root/entry['path']
        assert sha(checkpoint_path) == entry['sha256']
        price_checkpoint = read(checkpoint_path)
        first, last = date.fromisoformat(price_checkpoint['start']), date.fromisoformat(price_checkpoint['end'])
        days = {d for d in expected_dates if first.isoformat() <= d <= last.isoformat()}
        parts, verified = [], {}
        for object_id, source in catalog.items():
            if source['kind'] != 'normalized' or not first <= source['source_date'] <= last:
                continue
            normalized, frame = authenticated_frame(source)
            if frame.empty:
                raw_ids = [key for key, x in catalog.items() if x['kind'] == 'raw'
                           and x['payload_hash'] == normalized.payload_hash and x['source_date'] == normalized.source_date]
                assert len(raw_ids) == 1
                raw_id = raw_ids[0]
            else:
                assert frame.raw_object_id.nunique(dropna=False) == 1
                raw_id = str(frame.raw_object_id.iloc[0])
            assert raw_id in catalog and catalog[raw_id]['kind'] == 'raw'
            raw, original = authenticated_frame(catalog[raw_id])
            assert raw.payload_hash == normalized.payload_hash and raw.source_date == normalized.source_date
            left, right = frame.copy(), original.copy()
            left['date'] = pd.to_datetime(left.effective_date).dt.date
            right['date'] = pd.to_datetime(right.Date).dt.date
            # Match string storage dtypes as well as values: MultiIndex.equals
            # rejects object vs pandas StringDtype despite identical key tuples.
            left['provider_code'] = left.provider_code.astype(str)
            right['provider_code'] = right.Code.astype(str)
            for side in (left, right):
                assert side.date.eq(normalized.source_date).all()
                assert not side.duplicated(['date', 'provider_code']).any()
                assert side.provider_code.astype('string').str.fullmatch('[0-9A-Z]{5}').all()
            assert left.raw_object_id.eq(raw.object_id).all()
            assert left.symbol.eq(left.provider_code.str[:-1]).all()
            left = left.set_index(['date', 'provider_code']).sort_index()
            right = right.set_index(['date', 'provider_code']).sort_index()
            assert left.index.equals(right.index), 'Master raw/normalized keys differ'
            for column in _META:
                assert left[column].eq(right[column]).all(), column
            for target, origin in mapping.items():
                a, b = left[target].astype('string'), right[origin].astype('string')
                assert (a.eq(b) | (a.isna() & b.isna())).fillna(False).all(), target
            assert left.round_lot_size.isna().all(), 'Master does not provide trading lot sizes'
            assert left.round_lot_missing_reason.eq('not provided by /v2/equities/master').all()
            left['master_normalized_object_id'] = object_id
            left['master_raw_object_id'] = raw_id
            parts.append(left.reset_index())
            verified.update({normalized.object_id: normalized, raw.object_id: raw})
        assert parts, 'Historical master window has no authenticated source'
        members = pd.concat(parts, ignore_index=True)
        membership_keys = ['date', 'provider_code']
        priority = [*membership_keys, 'available_at', 'received_at', 'payload_hash']
        assert not members.duplicated(priority).any(), 'Ambiguous master revision priority'
        members = members.sort_values(priority).drop_duplicates(membership_keys, keep='last')
        members['date'] = members.date.map(date.isoformat)
        off_calendar_dates = sorted(set(members.date) - days)
        off_calendar_rows = int((~members.date.isin(days)).sum())
        members = members.loc[members.date.isin(days)].copy()
        missing_master_dates = sorted(days - set(members.date))
        members['eligible'] = members.provider_code.str.endswith('0') & members.market_code.isin(_ELIGIBLE_MARKETS)
        members['membership_state'] = 'ineligible_market'
        members.loc[~members.provider_code.str.endswith('0'), 'membership_state'] = 'nonstandard_provider_code'
        members.loc[members.eligible, 'membership_state'] = 'eligible_member'
        members.loc[~members.provider_code.str.endswith('0'), 'symbol'] = None
        price_path = root/price_checkpoint['receipt_path']
        price_bytes = price_path.read_bytes()
        assert hashlib.sha256(price_bytes).hexdigest() == price_checkpoint['receipt_sha256']
        prices = json.loads(price_bytes)
        assert prices['holdout_rows_read'] is False
        bars = pd.DataFrame(prices['rows'])[['trading_date', 'provider_code', 'bar_state',
                                           'normalized_object_id', 'raw_object_id', 'adjustment_requires_event_evidence']]
        bars = bars.rename(columns={'trading_date': 'date', 'normalized_object_id': 'price_normalized_object_id',
                                    'raw_object_id': 'price_raw_object_id'})
        assert set(bars.date) <= days and not bars.duplicated(membership_keys).any()
        members = members.rename(columns={'available_at': 'revision_available_at'})
        keep = [*membership_keys, 'symbol', 'market_code', 'sector_33_code', 'eligible', 'membership_state',
                'round_lot_size', 'revision_available_at', 'master_normalized_object_id', 'master_raw_object_id']
        joined = members[keep].merge(bars, on=membership_keys, how='outer', validate='one_to_one', indicator=True)
        joined.loc[joined._merge.eq('right_only'), 'membership_state'] = 'master_row_missing'
        joined.loc[joined._merge.eq('left_only'), 'bar_state'] = 'price_row_missing'
        joined = joined.drop(columns='_merge').sort_values(membership_keys).reset_index(drop=True)
        joined['eligible'] = joined.eligible.astype('boolean')
        profile = {}
        for day in sorted(days):
            rows = joined.loc[joined.date.eq(day)]
            eligible = rows.eligible.fillna(False)
            absent_price = rows.bar_state.eq('price_row_missing')
            values = {
                'union_rows': len(rows), 'master_rows': int(rows.master_normalized_object_id.notna().sum()),
                'price_rows': int(rows.price_normalized_object_id.notna().sum()),
                'eligible_rows': int(eligible.sum()), 'eligible_missing_price': int((eligible & absent_price).sum()),
                'eligible_reported_trade': int((eligible & rows.bar_state.eq('reported_trade')).sum()),
                'eligible_reported_no_trade': int((eligible & rows.bar_state.eq('reported_no_trade')).sum()),
                'eligible_incomplete_bar': int((eligible & rows.bar_state.eq('incomplete_bar')).sum()),
                'eligible_adjustment_rows': int((eligible & rows.adjustment_requires_event_evidence.fillna(False).astype(bool)).sum()),
                'price_missing_master': int(rows.membership_state.eq('master_row_missing').sum()),
                'all_master_missing_price': int(absent_price.sum()),
                'ineligible_market_rows': int(rows.membership_state.eq('ineligible_market').sum()),
                'nonstandard_provider_code_rows': int(rows.membership_state.eq('nonstandard_provider_code').sum()),

            assert values['eligible_rows'] == sum(values[k] for k in ['eligible_missing_price', 'eligible_reported_trade',
                    'eligible_reported_no_trade', 'eligible_incomplete_bar'])
            assert day not in daily
            profile[day] = values
            daily[day] = values

        union_path = output/f'window-{index:03d}-membership.parquet'
        joined.to_parquet(union_path, index=False)
        assert len(pd.read_parquet(union_path, columns=membership_keys)) == len(joined)
        for stored in verified.values():
            assert store.verify(stored.parquet_path.parent) == stored
        assert sha(price_path) == price_checkpoint['receipt_sha256']
        assert sha(checkpoint_path) == entry['sha256']
        all_verified.update(verified)
        result = {
            'window': index, 'start': first.isoformat(), 'end': last.isoformat(), 'per_day': profile,
            'missing_master_dates': missing_master_dates, 'off_calendar_master_dates': off_calendar_dates,
            'off_calendar_master_rows_not_joined': off_calendar_rows,
            'master_source_objects': [verified[key].model_dump(mode='json') for key in sorted(verified)],
            'price_receipt_path': str(price_path.relative_to(root)), 'price_receipt_sha256': sha(price_path),
            'projection_path': str(union_path.relative_to(root)), 'projection_sha256': sha(union_path),

        'all_source_files_unchanged': True, 'historical_as_known': False,
