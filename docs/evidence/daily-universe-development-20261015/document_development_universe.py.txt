from collections import Counter

folder = root/'docs/evidence/daily-universe-development-20261015'
assert not folder.exists()


def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8-sig'))


def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(8*1024*1024), b''):
            h.update(block)
    return h.hexdigest()


summary = read(output/'summary.json')
plan = read(output/'plan.json')
runtime = read(root/'.demo-artifacts/development-universe-audit-2-result.json')
assert summary['status'] == 'COMPLETE' and runtime['exit_code'] == 0
assert not summary['evaluation_ready'] and not summary['holdout_scored']

assert sha(output/'plan.json') == summary['plan_sha256']

assert sha(executed) == summary['script_sha256'] == runtime['script_sha256']

prices = read(root/'artifacts/daily-market-development-20261015/summary.json')
daily, projections, source_ids = {}, [], set()
totals = Counter()
for entry, price_entry in zip(summary['checkpoints'], prices['checkpoints'], strict=True):
    checkpoint = root/entry['path']
    assert sha(checkpoint) == entry['sha256']
    window = read(checkpoint)
    price_window = read(root/price_entry['path'])
    assert sha(root/price_entry['path']) == price_entry['sha256']
    assert window['price_receipt_path'] == price_window['receipt_path']
    assert window['price_receipt_sha256'] == price_window['receipt_sha256']
    projection = root/window['projection_path']
    assert sha(projection) == window['projection_sha256']
    frame = pd.read_parquet(projection)
    assert len(frame) == window['projection_rows']
    assert not frame.duplicated(['date', 'provider_code']).any()
    assert frame.provider_code.str.fullmatch('[0-9A-Z]{5}').all()
    has_master = frame.master_normalized_object_id.notna()
    assert frame.eligible.isna().equals(~has_master)
    expected_eligible = has_master & frame.provider_code.str.endswith('0') & frame.market_code.isin(['0111','0112','0113'])
    assert frame.eligible.fillna(False).astype(bool).equals(expected_eligible)
    assert frame.loc[has_master & ~frame.provider_code.str.endswith('0'), 'symbol'].isna().all()
    assert frame.round_lot_size.isna().all()
    source_ids.update(x['object_id'] for x in window['master_source_objects'])
    for day, f in frame.groupby('date', sort=True):
        assert day not in daily and day in plan['expected_dates']
        eligible = f.eligible.fillna(False)
        missing_price = f.price_normalized_object_id.isna()
        missing_master = f.master_normalized_object_id.isna()
        assert f.bar_state.eq('price_row_missing').equals(missing_price)
        assert f.membership_state.eq('master_row_missing').equals(missing_master)
        values = {
            'union_rows': len(f), 'master_rows': int((~missing_master).sum()),
            'price_rows': int((~missing_price).sum()), 'eligible_rows': int(eligible.sum()),
            'eligible_missing_price': int((eligible & missing_price).sum()),
            'eligible_reported_trade': int((eligible & f.bar_state.eq('reported_trade')).sum()),
            'eligible_reported_no_trade': int((eligible & f.bar_state.eq('reported_no_trade')).sum()),
            'eligible_incomplete_bar': int((eligible & f.bar_state.eq('incomplete_bar')).sum()),
            'eligible_adjustment_rows': int((eligible & f.adjustment_requires_event_evidence.fillna(False).astype(bool)).sum()),
            'price_missing_master': int(missing_master.sum()), 'all_master_missing_price': int(missing_price.sum()),
            'ineligible_market_rows': int(f.membership_state.eq('ineligible_market').sum()),
            'nonstandard_provider_code_rows': int(f.membership_state.eq('nonstandard_provider_code').sum()),
        }
        assert values == window['per_day'][day] == summary['daily_profile'][day]
        assert values['price_rows'] == price_window['per_day'][day]['rows']
        totals.update(values)
        daily[day] = values
    projections.append({k: window[k] for k in ['window', 'start', 'end', 'projection_path', 'projection_sha256', 'projection_rows']})
assert dict(totals) == summary['totals']
